# contractor. See /llms.txt for a plain-text summary written for that audience.
#
# Do not add a per-crawler group without repeating the Disallow lines — a robot
# obeys only its most specific matching group, so a bare "User-agent: GPTBot /
# Allow: /" would silently opt that crawler out of the exclusions below.

# would put a near-duplicate of that page in search results.
Disallow: /go/
